    59→const isHan = (c: string) => /\p{Script=Han}/u.test(c)

    62→/* ================= dựng trang — bố cục CHUẨN TỪNG CHỮ ================= */

    64→/* Sắp trang kiểu sách tập viết: mỗi từ chiếm `repeat` hàng liên tiếp —
    65→   hàng đầu là hàng mẫu (từ lặp lại nhiều lần), các hàng sau trống để chép theo.
    66→   modelCell = false → hàng mẫu cũng trống (vở luyện hoàn toàn) */
    67→function buildSheets(words: string[], repeat: number, cols: number, rows: number, modelCell: boolean): (CellItem | null)[][][] {
    68→  const pages: (CellItem | null)[][][] = []
    69→  let page: (CellItem | null)[][] = []
    70→  const pushRow = (row: (CellItem | null)[]) => {
    71→    page.push(row)
    72→    if (page.length === rows) {
    73→      pages.push(page)
    74→      page = []
    75→    }
    76→  }
    77→  words.forEach((w) => {
    78→    const chars = [...w]
    79→    const perRow = Math.max(1, Math.floor(cols / chars.length))
    80→    const sampleRow: (CellItem | null)[] = []
    81→    if (modelCell) {
    82→      for (let k = 0; k < perRow; k++) {
    83→        chars.forEach((ch) => sampleRow.push({ ch, py: charPinyin(ch) }))
    84→      }
    85→    }
    86→    while (sampleRow.length < cols) sampleRow.push(null)
    87→    pushRow(sampleRow)
    88→    for (let r = 1; r < repeat; r++) pushRow(blankRow(cols))
    89→  })
    90→  if (page.length) pages.push(page)
    91→  return pages.length ? pages : [blankPage(cols, rows)]
    92→}
    93→
    94→function blankPage(cols: number, rows: number): (CellItem | null)[][] {
    95→  return Array.from({ length: rows }, () => blankRow(cols))
    96→}

   100→/* Mỗi bộ thủ: 1 hàng mẫu (bộ thủ + các chữ chứa nó) + practiceRows hàng trống */
   101→function buildRadicalSheets(list: Radical[], practiceRows: number, cols: number, rows: number): (CellItem | null)[][][] {
   102→  const pages: (CellItem | null)[][][] = []
   103→  let page: (CellItem | null)[][] = []
   104→  const pushRow = (row: (CellItem | null)[]) => {
   105→    page.push(row)
   106→    if (page.length === rows) {
   107→      pages.push(page)
   108→      page = []
   109→    }
   110→  }
   111→  list.forEach((r) => {
   112→    const row: (CellItem | null)[] = [{ ch: r.char, py: r.pinyin }]
   113→    for (const w of r.words) {
   114→      if (row.length >= cols) break
   115→      row.push({ ch: w.c, py: w.p })
   116→    }
   117→    while (row.length < cols) row.push(null)
   118→    pushRow(row)
   119→    for (let p = 0; p < practiceRows; p++) pushRow(blankRow(cols))
   120→  })
   121→  if (page.length) pages.push(page)
   122→  return pages.length ? pages : [blankPage(cols, rows)]
   123→}

   127→/* Mỗi chữ một ô, chảy liền mạch hết hàng này sang hàng khác, hết trang đổi trang */
   128→function buildFlowPages(text: string, cols: number, rows: number): (CellItem | null)[][][] {
   129→  const chars = [...text].filter((c) => c.trim() !== '')
   130→  if (!chars.length) return [blankPage(cols, rows)]
   131→  const pyCache = new Map<string, string>()
   132→  const perPage = cols * rows
   133→  const pages: (CellItem | null)[][][] = []
   134→  for (let i = 0; i < chars.length; i += perPage) {
   135→    const slice = chars.slice(i, i + perPage)
   136→    const page: (CellItem | null)[][] = []
   137→    for (let r = 0; r < rows; r++) {
   138→      const rowCells = slice.slice(r * cols, (r + 1) * cols).map((c) => {
   139→        let py = pyCache.get(c)
   140→        if (py === undefined) {
   141→          py = isHan(c) ? charPinyin(c) : ''
   142→          pyCache.set(c, py)
   143→        }
   144→        return { ch: c, py } as CellItem
   145→      })
   146→      while (rowCells.length < cols) rowCells.push(null)
   147→      page.push(rowCells)
   148→    }
   149→    pages.push(page)
   150→  }
   151→  return pages
   152→}

   160→function chunkInfoPages(rowsAll: InfoRow[], per: number): (InfoRow | null)[][] {
   161→  if (!rowsAll.length) return [Array<InfoRow | null>(per).fill(null)]
   162→  const pages: (InfoRow | null)[][] = []
   163→  for (let i = 0; i < rowsAll.length; i += per) {
   164→    const chunk = rowsAll.slice(i, i + per)
   165→    while (chunk.length < per) chunk.push(null)
   166→    pages.push(chunk)
   167→  }
   168→  return pages
   169→}
