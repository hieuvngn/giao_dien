    76→const isHan = (c: string) => /\p{Script=Han}/u.test(c)

    80→/* ================= dựng trang — bố cục CHUẨN TỪNG CHỮ ================= */
    81→
    82→/* Sắp trang kiểu sách tập viết: mỗi từ chiếm `repeat` hàng liên tiếp —
    83→   hàng đầu là hàng mẫu (từ lặp lại nhiều lần), các hàng sau trống để chép theo.
    84→   modelCell = false → hàng mẫu cũng trống (vở luyện hoàn toàn) */
    85→function buildSheets(words: string[], repeat: number, cols: number, rows: number, modelCell: boolean): (CellItem | null)[][][] {
    86→  const pages: (CellItem | null)[][][] = []
    87→  let page: (CellItem | null)[][] = []
    88→  const pushRow = (row: (CellItem | null)[]) => {
    89→    page.push(row)
    90→    if (page.length === rows) {
    91→      pages.push(page)
    92→      page = []
    93→    }
    94→  }
    95→  words.forEach((w) => {
    96→    const chars = [...w]
    97→    const perRow = Math.max(1, Math.floor(cols / chars.length))
    98→    const sampleRow: (CellItem | null)[] = []
    99→    if (modelCell) {
   100→      for (let k = 0; k < perRow; k++) {
   101→        chars.forEach((ch) => sampleRow.push({ ch, py: charPinyin(ch) }))
   102→      }
   103→    }
   104→    while (sampleRow.length < cols) sampleRow.push(null)
   105→    pushRow(sampleRow)
   106→    for (let r = 1; r < repeat; r++) pushRow(blankRow(cols))
   107→  })
   108→  if (page.length) pages.push(page)
   109→  return pages.length ? pages : [blankPage(cols, rows)]
   110→}
   111→
   112→function blankPage(cols: number, rows: number): (CellItem | null)[][] {
   113→  return Array.from({ length: rows }, () => blankRow(cols))
   114→}

   118→/* Mỗi bộ thủ: 1 hàng mẫu (bộ thủ + các chữ chứa nó) + practiceRows hàng trống */
   119→function buildRadicalSheets(list: Radical[], practiceRows: number, cols: number, rows: number): (CellItem | null)[][][] {
   120→  const pages: (CellItem | null)[][][] = []
   121→  let page: (CellItem | null)[][] = []
   122→  const pushRow = (row: (CellItem | null)[]) => {
   123→    page.push(row)
   124→    if (page.length === rows) {
   125→      pages.push(page)
   126→      page = []
   127→    }
   128→  }
   129→  list.forEach((r) => {
   130→    const row: (CellItem | null)[] = [{ ch: r.char, py: r.pinyin }]
   131→    for (const w of r.words) {
   132→      if (row.length >= cols) break
   133→      row.push({ ch: w.c, py: w.p })
   134→    }
   135→    while (row.length < cols) row.push(null)
   136→    pushRow(row)
   137→    for (let p = 0; p < practiceRows; p++) pushRow(blankRow(cols))
   138→  })
   139→  if (page.length) pages.push(page)
   140→  return pages.length ? pages : [blankPage(cols, rows)]
   141→}

   143→/* ================= dựng trang — bố cục VĂN BẢN LIÊN TỤC ================= */

   145→/* Mỗi chữ một ô, chảy liền mạch hết hàng này sang hàng khác, hết trang đổi trang */
   146→function buildFlowPages(text: string, cols: number, rows: number): (CellItem | null)[][][] {
   147→  const chars = [...text].filter((c) => c.trim() !== '')
   148→  if (!chars.length) return [blankPage(cols, rows)]
   149→  const pyCache = new Map<string, string>()
   150→  const perPage = cols * rows
   151→  const pages: (CellItem | null)[][][] = []
   152→  for (let i = 0; i < chars.length; i += perPage) {
   153→    const slice = chars.slice(i, i + perPage)
   154→    const page: (CellItem | null)[][] = []
   155→    for (let r = 0; r < rows; r++) {
   156→      const rowCells = slice.slice(r * cols, (r + 1) * cols).map((c) => {
   157→        let py = pyCache.get(c)
   158→        if (py === undefined) {
   159→          py = isHan(c) ? charPinyin(c) : ''
   160→          pyCache.set(c, py)
   161→        }
   162→        return { ch: c, py } as CellItem
   163→      })
   164→      while (rowCells.length < cols) rowCells.push(null)
   165→      page.push(rowCells)
   166→    }
   167→    pages.push(page)
   168→  }
   169→  return pages
   170→}

   172→/* ================= dựng trang — bố cục THỨ TỰ NÉT ================= */
   173→
   174→/* Mỗi từ 1 khối: dòng tiêu đề (từ · pinyin · nghĩa · sơ đồ thứ tự nét)
   175→   + 1 hàng chữ đỏ lặp đầy hàng + các hàng chữ mờ để tô theo.          */
   176→function buildStrokePages(words: string[], repeat: number, cols: number, rows: number): StrokeBlock[][] {
   177→  const blocks: StrokeBlock[] = words.map((w) => {
   178→    const chars = [...w]
   179→    const py = wordPinyin(w)
   180→    const vi = MEANING_MAP.get(w) ?? ''
   181→    const modelRow: (CellItem | null)[] = []
   182→    for (let c = 0; c < cols; c++) modelRow.push({ ch: chars[c % chars.length], py: charPinyin(chars[c % chars.length]), v: 'model' })
   183→    const rowsAll: (CellItem | null)[][] = [modelRow]
   184→    for (let r = 0; r < Math.max(0, repeat - 1); r++) {
   185→      const row: (CellItem | null)[] = []
   186→      for (let c = 0; c < cols; c++) row.push({ ch: chars[c % chars.length], py: '', v: 'trace' })
   187→      rowsAll.push(row)
   188→    }
   189→    return { word: w, py, vi, rows: rowsAll }
   190→  })
   191→  const perPage = Math.max(1, Math.floor((rows + 0.55) / (repeat + 0.45)))
   192→  const pages: StrokeBlock[][] = []
   193→  for (let i = 0; i < blocks.length; i += perPage) pages.push(blocks.slice(i, i + perPage))
   194→  return pages.length ? pages : [[]]
   195→}
   196→
   197→/* ================= dựng trang — bố cục Ô CHỮ LỚN ================= */
   198→
   199→/* Tách từng ký tự: mỗi chữ 1 dải — chữ mẫu to + pinyin + nghĩa,
   200→   bên dưới là hàng ô chữ mờ để tô.                              */
   201→function buildBigPages(words: string[], rows: number): BigBand[][] {
   202→  const bands: BigBand[] = []
   203→  const seen = new Set<string>()
   204→  words.forEach((w) =>
   205→    [...w].forEach((c) => {
   206→      if (!isHan(c) || seen.has(c)) return
   207→      seen.add(c)
   208→      bands.push({ ch: c, py: charPinyin(c), vi: MEANING_MAP.get(c) ?? radicalMeaning(c) })
   209→    }),
   210→  )
   211→  const perPage = Math.max(1, Math.floor(rows / 2))
   212→  const pages: BigBand[][] = []
   213→  for (let i = 0; i < bands.length; i += perPage) pages.push(bands.slice(i, i + perPage))
   214→  return pages.length ? pages : [[]]
   215→}

   217→/* ================= dựng trang — bố cục BẢNG THÔNG TIN ================= */
   218→
   219→function buildInfoRows(words: string[]): InfoRow[] {
   220→  return words.map((w, i) => ({ n: i + 1, cn: w, py: wordPinyin(w), vi: MEANING_MAP.get(w) ?? '' }))
   221→}
   222→
   223→function chunkInfoPages(rowsAll: InfoRow[], per: number): (InfoRow | null)[][] {
   224→  if (!rowsAll.length) return [Array<InfoRow | null>(per).fill(null)]
   225→  const pages: (InfoRow | null)[][] = []
   226→  for (let i = 0; i < rowsAll.length; i += per) {
   227→    const chunk = rowsAll.slice(i, i + per)
   228→    while (chunk.length < per) chunk.push(null)
   229→    pages.push(chunk)
   230→  }
   231→  return pages
   232→}
