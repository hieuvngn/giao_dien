    39→function lsGet<T>(key: string, fallback: T): T {
    40→  try {
    41→    const raw = localStorage.getItem(key)
    42→    return raw ? (JSON.parse(raw) as T) : fallback
    43→  } catch {
    44→    return fallback
    45→  }
    46→}
    47→function lsSet(key: string, value: unknown) {
    48→  try {
    49→    localStorage.setItem(key, JSON.stringify(value))
    50→  } catch {
    51→    /* bỏ qua */
    52→  }
    53→}

    60→function speak(text: string) {
    61→  try {
    62→    if (typeof window === 'undefined' || !('speechSynthesis' in window)) return
    63→    const u = new SpeechSynthesisUtterance(text)
    64→    u.lang = 'zh-CN'
    65→    u.rate = 0.85
    66→    const voice = window.speechSynthesis.getVoices().find((v) => v.lang.toLowerCase().startsWith('zh'))
    67→    if (voice) u.voice = voice
    68→    window.speechSynthesis.cancel()
    69→    window.speechSynthesis.speak(u)
    70→  } catch {
    71→    /* im lặng */
    72→  }
    73→}

   285→export default function LessonStudy({ slug, kind, n }: { slug?: string; kind?: Kind; n?: number }) {
   286→  const { toast } = useToast()
   287→  const level = findRoadmapLevel(slug)
   288→  const k: Kind = kind === 'g' ? 'g' : 'v'
   289→  const num = Number(n) || 1
   290→
   291→  const vocabLesson = level?.vocab.find((l) => l.n === num)
   292→  const grammarLesson = level?.grammar.find((l) => l.n === num)
   293→
   294→  /* ----- trạng thái học ----- */
   295→  const [showPinyin, setShowPinyin] = useState(true)
   296→  const [autoSpeak, setAutoSpeak] = useState(false)
   297→  const [tab, setTab] = useState<'card' | 'example'>('card')
   298→  const [front, setFront] = useState<'zh' | 'vi'>('zh')
   299→  const [auto, setAuto] = useState(false)
   300→  const [idx, setIdx] = useState(0)
   301→  const [shuffled, setShuffled] = useState<{ key: string; order: number[] } | null>(null)
   302→  const [settingsOpen, setSettingsOpen] = useState(false)
   303→  const [meta, setMeta] = useState<Meta>(EMPTY_META)
   304→  const countedRef = useRef(false)
   305→
   306→  const deckKey = `${k}:${num}:${tab}`
   307→
   308→  /* ----- bộ thẻ ----- */
   309→  const cards: StudyCard[] = useMemo(() => {
   310→    if (!level) return []
   311→    if (tab === 'example') {
   312→      if (k === 'v') {
   313→        return vocabLesson
   314→          ? vocabLesson.words.filter((w) => w.ex).map((w) => ({ cn: w.ex!.cn, pinyin: w.ex!.pinyin, vi: w.ex!.vi, speak: w.ex!.cn }))
   315→          : []
   316→      }
   317→      return grammarLesson
   318→        ? [{ cn: grammarLesson.example.cn, pinyin: grammarLesson.example.pinyin, vi: grammarLesson.example.vi, speak: grammarLesson.example.cn }]
   319→        : []
   320→    }
   321→    if (k === 'v') {
   322→      return vocabLesson
   323→        ? vocabLesson.words.map((w) => ({ cn: w.cn, pinyin: w.pinyin, vi: w.vi, pos: w.pos, speak: w.cn }))
   324→        : []
   325→    }
   326→    return grammarLesson
   327→      ? [{ cn: grammarLesson.structure, pinyin: grammarLesson.cn ?? '', vi: grammarLesson.point, example: grammarLesson.example, speak: grammarLesson.example.cn }]
   328→      : []
   329→  }, [level, k, tab, vocabLesson, grammarLesson])
   330→
   331→  /* thứ tự thẻ: dùng thứ tự xáo trộn nếu còn hợp lệ, ngược lại theo thứ tự gốc
   332→     (đổi tab/bài sẽ tự về thứ tự gốc vì key của trạng thái xáo trộn không còn khớp) */
   333→  const order = useMemo(() => {
   334→    if (shuffled && shuffled.key === deckKey && shuffled.order.length === cards.length) return shuffled.order
   335→    return cards.map((_, i) => i)
   336→  }, [shuffled, deckKey, cards])
   337→
   338→  const total = cards.length
   339→  const safeIdx = Math.min(idx, Math.max(total - 1, 0))
   340→  const cardIndex = tab === 'card' ? order[safeIdx] ?? safeIdx : safeIdx
   341→  const card: StudyCard | undefined = cards[safeIdx]
   342→
   343→  /* ----- khôi phục trạng thái từ localStorage + đếm lượt học ----- */
   344→  useEffect(() => {
   345→    if (!level) return
   346→    const studies = lsGet<Record<string, number>>('nkhn.studies', {})
   347→    if (!countedRef.current) {
   348→      countedRef.current = true
   349→      const key = studyKey(level.slug, k, num)
   350→      studies[key] = (studies[key] ?? 0) + 1
   351→      lsSet('nkhn.studies', studies)
   352→    }
   353→    // eslint-disable-next-line react-hooks/set-state-in-effect
   354→    setMeta({
   355→      stars: lsGet('nkhn.stars', {}),
   356→      pins: lsGet('nkhn.pins', {}),
   357→      known: lsGet(knownKey(level.slug, k, num), {}),
   358→      studyCount: studies[studyKey(level.slug, k, num)] ?? 0,
   359→    })
   360→  }, [level, k, num])
   361→
   362→  const shuffle = useCallback(() => {
   363→    const arr = cards.map((_, i) => i)
   364→    for (let i = arr.length - 1; i > 0; i--) {
   365→      const j = Math.floor(Math.random() * (i + 1))
   366→      ;[arr[i], arr[j]] = [arr[j], arr[i]]
   367→    }
   368→    setShuffled({ key: deckKey, order: arr })
   369→    setIdx(0)
   370→    toast({ title: '🔀 Đã xáo trộn bộ thẻ!' })
   371→  }, [cards, deckKey, toast])
   372→
   373→  /* ----- điều hướng thẻ ----- */
   374→  const go = useCallback(
   375→    (d: 1 | -1) => {
   376→      setIdx((i) => {
   377→        if (total === 0) return 0
   378→        const s = Math.min(i, total - 1)
   379→        if (d === 1) return s + 1 >= total ? 0 : s + 1
   380→        return s - 1 < 0 ? total - 1 : s - 1
   381→      })
   382→    },
   383→    [total],
   384→  )
   385→
   386→  const next = useCallback(() => go(1), [go])
   387→
   388→  /* tự động chuyển thẻ */
   389→  useEffect(() => {
   390→    if (!auto || total === 0) return
   391→    const t = setInterval(next, 3200)
   392→    return () => clearInterval(t)
   393→  }, [auto, next, total])
   394→
   395→  /* đánh dấu thuộc / chưa thuộc */
   396→  const mark = useCallback(
   397→    (m: 'known' | 'unknown') => {
   398→      if (!level || total === 0) return
   399→      const id = `${k}:${num}:${cardIndex}`
   400→      setMeta((prev) => {
   401→        const nextMeta = { ...prev, known: { ...prev.known, [id]: m } }
   402→        lsSet(knownKey(level.slug, k, num), nextMeta.known)
   403→        return nextMeta
   404→      })
   405→      if (m === 'known') toast({ title: '✓ Giỏi lắm! Đã thuộc thẻ này.' })
   406→      else toast({ title: '✗ Không sao, lần sau nhớ nhé!' })
   407→      next()
   408→    },
   409→    [level, total, cardIndex, k, num, next, toast],
   410→  )
   411→
   412→  /* bàn phím */
   413→  useEffect(() => {
   414→    const onKey = (e: KeyboardEvent) => {
   415→      if (e.target instanceof HTMLElement && ['INPUT', 'TEXTAREA', 'SELECT'].includes(e.target.tagName)) return
   416→      if (e.key === 'ArrowRight') next()
   417→      else if (e.key === 'ArrowLeft') go(-1)
   418→      else if (e.key === ' ') {
   419→        e.preventDefault()
   420→        document.querySelector<HTMLElement>('[aria-label="Lật thẻ"]')?.click()
   421→      } else if (e.key === '1') mark('unknown')
   422→      else if (e.key === '2') mark('known')
   423→    }
   424→    document.addEventListener('keydown', onKey)
   425→    return () => document.removeEventListener('keydown', onKey)
   426→  }, [next, go, mark])
   427→
   428→  /* toàn màn hình */
   429→  const toggleFullscreen = () => {
   430→    try {
   431→      if (document.fullscreenElement) void document.exitFullscreen()
   432→      else void document.documentElement.requestFullscreen()
   433→    } catch {
   434→      /* bỏ qua */
   435→    }
   436→  }
   437→
   438→  /* ghim / sao */
   439→  const toggleStar = (id: string) => {
   440→    setMeta((prev) => {
   441→      const nextMeta = { ...prev, stars: { ...prev.stars, [id]: !prev.stars[id] } }
   442→      lsSet('nkhn.stars', nextMeta.stars)
   443→      return nextMeta
   444→    })
   445→  }
   446→  const togglePin = (id: string) => {
   447→    const wasPinned = !!meta.pins[id]
   448→    setMeta((prev) => {
   449→      const nextMeta = { ...prev, pins: { ...prev.pins, [id]: !prev.pins[id] } }
   450→      lsSet('nkhn.pins', nextMeta.pins)
   451→      return nextMeta
