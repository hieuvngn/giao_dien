    59→    59→const isHan = (c: string) => /\p{Script=Han}/u.test(c)
    60→    60→const blankRow = (cols: number): (CellItem | null)[] => Array<CellItem | null>(cols).fill(null)

    62→    62→/* ================= dựng trang — bố cục CHUẨN TỪNG CHỮ ================= */

    64→    64→/* Sắp trang kiểu sách tập viết: mỗi từ chiếm `repeat` hàng liên tiếp —
    65→    65→   hàng đầu là hàng mẫu (từ lặp lại nhiều lần), các hàng sau trống để chép theo.
    66→    66→   modelCell = false → hàng mẫu cũng trống (vở luyện hoàn toàn) */
    67→    67→function buildSheets(words: string[], repeat: number, cols: number, rows: number, modelCell: boolean): (CellItem | null)[][][] {
    68→    68→  const pages: (CellItem | null)[][][] = []
    69→    69→  let page: (CellItem | null)[][] = []
    70→    70→  const pushRow = (row: (CellItem | null)[]) => {
    71→    71→    page.push(row)
    72→    72→    if (page.length === rows) {
    73→    73→      pages.push(page)
    74→    74→      page = []
    75→    75→    }
    76→    76→  }
    77→    77→  words.forEach((w) => {
    78→    78→    const chars = [...w]
    79→    79→    const perRow = Math.max(1, Math.floor(cols / chars.length))
    80→    80→    const sampleRow: (CellItem | null)[] = []
    81→    81→    if (modelCell) {
    82→    82→      for (let k = 0; k < perRow; k++) {
    83→    83→        chars.forEach((ch) => sampleRow.push({ ch, py: charPinyin(ch) }))
    84→    84→      }
    85→    85→    }
    86→    86→    while (sampleRow.length < cols) sampleRow.push(null)
    87→    87→    pushRow(sampleRow)
    88→    88→    for (let r = 1; r < repeat; r++) pushRow(blankRow(cols))
    89→    89→  })
    90→    90→  if (page.length) pages.push(page)
    91→    91→  return pages.length ? pages : [blankPage(cols, rows)]
    92→    92→}
    93→    93→
    94→    94→function blankPage(cols: number, rows: number): (CellItem | null)[][] {
    95→    95→  return Array.from({ length: rows }, () => blankRow(cols))
    96→    96→}

   100→   100→/* Mỗi bộ thủ: 1 hàng mẫu (bộ thủ + các chữ chứa nó) + practiceRows hàng trống */
   101→   101→function buildRadicalSheets(list: Radical[], practiceRows: number, cols: number, rows: number): (CellItem | null)[][][] {
   102→   102→  const pages: (CellItem | null)[][][] = []
   103→   103→  let page: (CellItem | null)[][] = []
   104→   104→  const pushRow = (row: (CellItem | null)[]) => {
   105→   105→    page.push(row)
   106→   106→    if (page.length === rows) {
   107→   107→      pages.push(page)
   108→   108→      page = []
   109→   109→    }
   110→   110→  }
   111→   111→  list.forEach((r) => {
   112→   112→    const row: (CellItem | null)[] = [{ ch: r.char, py: r.pinyin }]
   113→   113→    for (const w of r.words) {
   114→   114→      if (row.length >= cols) break
   115→   115→      row.push({ ch: w.c, py: w.p })
   116→   116→    }
   117→   117→    while (row.length < cols) row.push(null)
   118→   118→    pushRow(row)
   119→   119→    for (let p = 0; p < practiceRows; p++) pushRow(blankRow(cols))
   120→   120→  })
   121→   121→  if (page.length) pages.push(page)
   122→   122→  return pages.length ? pages : [blankPage(cols, rows)]
   123→   123→}

   125→   125→/* ================= dựng trang — bố cục VĂN BẢN LIÊN TỤC ================= */
   126→   126→
   127→   127→/* Mỗi chữ một ô, chảy liền mạch hết hàng này sang hàng khác, hết trang đổi trang */
   128→   128→function buildFlowPages(text: string, cols: number, rows: number): (CellItem | null)[][][] {
   129→   129→  const chars = [...text].filter((c) => c.trim() !== '')
   130→   130→  if (!chars.length) return [blankPage(cols, rows)]
   131→   131→  const pyCache = new Map<string, string>()
   132→   132→  const perPage = cols * rows
   133→   133→  const pages: (CellItem | null)[][][] = []
   134→   134→  for (let i = 0; i < chars.length; i += perPage) {
   135→   135→    const slice = chars.slice(i, i + perPage)
   136→   136→    const page: (CellItem | null)[][] = []
   137→   137→    for (let r = 0; r < rows; r++) {
   138→   138→      const rowCells = slice.slice(r * cols, (r + 1) * cols).map((c) => {
   139→   139→        let py = pyCache.get(c)
   140→   140→        if (py === undefined) {
   141→   141→          py = isHan(c) ? charPinyin(c) : ''
   142→   142→          pyCache.set(c, py)
   143→   143→        }
   144→   144→        return { ch: c, py } as CellItem
   145→   145→      })
   146→   146→      while (rowCells.length < cols) rowCells.push(null)
   147→   147→      page.push(rowCells)
   148→   148→    }
   149→   149→    pages.push(page)
   150→   150→  }
   151→   151→  return pages
   152→   152→}

   156→   156→function buildInfoRows(words: string[]): InfoRow[] {
   157→   157→  return words.map((w, i) => ({ n: i + 1, cn: w, py: wordPinyin(w), vi: MEANING_MAP.get(w) ?? '' }))

   159→   159→
   160→   160→function chunkInfoPages(rowsAll: InfoRow[], per: number): (InfoRow | null)[][] {
   161→   161→  if (!rowsAll.length) return [Array<InfoRow | null>(per).fill(null)]
   162→   162→  const pages: (InfoRow | null)[][] = []
   163→   163→  for (let i = 0; i < rowsAll.length; i += per) {
   164→   164→    const chunk = rowsAll.slice(i, i + per)
   165→   165→    while (chunk.length < per) chunk.push(null)
   166→   166→    pages.push(chunk)
   167→   167→  }
   168→   168→  return pages
   169→   169→}
